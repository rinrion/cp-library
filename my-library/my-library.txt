// 隣接グラフ startからgoalへの最短ステップ数				
int bfsDistGraph(int n, int start, int goal, vvi &g) {				
	vi dist(n, -1);			
	queue<int> q;			
				
	dist[start] = 0;			
	q.push(start);			
				
	while (!q.empty()) {			
		int now = q.front();		
		q.pop();		
				
		if (now == goal) return dist[now];		
				
		for (auto next : g[now]) {		
			if (dist[next] != -1) continue;	
			dist[next] = dist[now] + 1;	
			q.push(next);	
		}		
	}			
	return -1;  // 到達不可			
}

		}		
	}

// 隣接グラフ 各頂点にたどり着けるか 1: 到達 0: 未達				
vi bfsReachGraph (int n, int start, vvi g){				
	vi isReach (n, 0);			
	queue<int> q;			
				
	isReach[start] = 1; // start			
	q.push (start);			
				
	while (!q.empty()){			
		int now = q.front ();		
		q.pop ();		
		for (auto next : g[now]){		
			if (isReach[next] != 0) continue;	
			isReach[next] = 1;	
			q.push (next);	
		}		
	}			
	return isReach;			
}

// goalに到達できない場合はisReachdがfalseのまま				
bool isReachd = false;				
vi bfsRouteGraph (int n, int start, int goal, vvi g, bool &isReachd){				
	vi route; // 経路を入れる配列			
	vi route_before (n, -2); // 直前にいた頂点			
	queue<int> q;			
	route_before[start] = -1;			
	q.push (start);			
				
	while (!q.empty ()){			
		int now = q.front ();		
		q.pop ();		
				
		for (auto next : g[now]){		
			if (route_before[next] != -2) continue; // すでに訪問済み	
			route_before[next] = now; // 直前にいた頂点を記録	
				
			if (next == goal) {	
				isReachd = true;
				break; // goalに到達したら終了
			}	
			q.push (next);	
		}		
		if (isReachd) break;		
	}			
	if (!isReachd) return route; // goalに到達できない場合			
				
	int now = goal; // 頂点goalまでの経路をrouteに追加→reverse			
	while (now != -1) {			
		route.push_back (now);		
		now = route_before[now];		
	}			
				
	reverse (route.begin (), route.end ());			
	return route;			
}

vi bfsDirGrid (int h, int w, point start, point goal, bool &isReachd){				
	vi route;			
	vector<vector<pair<point, int>>> route_before (h, vector<pair<point, int>>(w, {{-2, -2}, -1})); // 直前のマス, 来た方向			
	queue<point> q;			
	route_before[start.i][start.j].first = {-1, -1};			
	q.push (start);			
				
	while (!q.empty ()){			
		point now = q.front ();		
		q.pop ();		
				
		rep (dir, 0, 4){		
			point next = getMoveCell(now, dir, 1, h, w);	
			if(next.i == -1) continue; // 枠外	
			if (route_before[next.i][next.j].second != -1) continue; // 来た方向が記録済:すでに訪問済み	
			if (route_before[next.i][next.j].first.i != -2) continue; // 直前のマスが記録済:すでに訪問済み	
				
			route_before[next.i][next.j] = make_pair (point{now.i, now.j}, dir); // 直前のマス, 来た方向を記録	
			if (now.i == goal.i && now.j == goal.j){	
				isReachd = true; // goalに到達したら終了
				break;
			}	
			q.push (next);	
		}		
		if (isReachd) break;		
	}			
	if (!isReachd) return route; // goalに到達できない場合			
				
	point now = goal;			
	while (now != start){			
		route.push_back (route_before[now.i][now.j].second);		
		now = route_before[now.i][now.j].first;		
	}			
				
	reverse (route.begin (), route.end ());			
	return route;			
}

void dfs(int p){
  if(seen[p]) return;
  seen[p] = true;

  for(/*次の状態の候補集合*/){
    dfs(/*次の状態*/);
  }
}

void dfs(int v){
  if(seen[v]) return;
  ans += t[v];
  seen[v] = 1;
  for (auto x : a[v]) dfs(x);

void dfs(/* 今選んでいるもの,合計値等の変数*/) {
  if (/* 終了状態*/) {
    /* 答えを更新*/
    return;
  }

  for (/* 候補につき*/) {
    if (/* 選べない候補なら*/) continue;

    // choose seen=1など
    dfs(/* 次の状態*/);
    // unchoose seen=0など
  }
}

int binary_search(int key) {
  //「index = 0」が条件を満たすこともあるので、初期値は -1
  int ng = -1; 
  // 最大値が条件を満たさないこともあるので、初期値は 最大値＋α
  int ok =/*最大値*/+α; 

  while (ok - ng > 1) {
    int mid = ng + (ok - ng) / 2;
    if (mid >= key) ok = mid;
    else ng = mid;
  }  
  return ok;
}
